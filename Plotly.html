instacart_df = 
  instacart |> 
  janitor::clean_names() |>
  group_by(
    product_id, product_name
  ) |> 
  summarize(
    n_purchases = n(),
    mean_cart_position = mean(add_to_cart_order),
    reorder_rate = mean(reordered),
    n_aisles = n_distinct(aisle)
  ) |> 
  ungroup()
